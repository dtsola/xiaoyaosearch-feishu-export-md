```typescript
import type { TenantAccessTokenResponse } from '../types/index.js';

/**
 * 认证管理器
 * 负责获取和缓存 tenant_access_token
 */
export class AuthManager {
  private accessToken: string | null = null;
  private expiresAt: number = 0;
  private cacheFile: string;

  constructor(
    private endpoint: string,
    private appId: string,
    private appSecret: string,
    cacheDir: string = '~/.feishu-export'
  ) {
    this.cacheFile = `${cacheDir}/cache.json`;
  }

  /**
   * 获取 access token
   * 自动处理缓存和刷新
   */
  async getAccessToken(): Promise<string> {
    // 检查缓存
    if (this.accessToken && Date.now() < this.expiresAt) {
      return this.accessToken;
    }

    // 获取新 token
    await this.refreshToken();
    return this.accessToken!;
  }

  /**
   * 刷新 access token
   */
  private async refreshToken(): Promise<void> {
    const url = `${this.endpoint}/open-apis/auth/v3/tenant_access_token/internal`;

    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        app_id: this.appId,
        app_secret: this.appSecret,
      }),
    });

    if (!response.ok) {
      throw new Error(`认证失败: HTTP ${response.status}`);
    }

    const data = (await response.json()) as TenantAccessTokenResponse;

    if (data.code !== 0) {
      throw new Error(`认证失败: [${data.code}] ${data.msg}`);
    }

    this.accessToken = data.tenant_access_token;
    // 提前 5 分钟过期
```

```typescript
import type {
  DocumentInfo,
  DocumentBlocksResponse,
  WikiNodeInfo,
  Block,
  FeishuApiResponse,
} from '../types/index.js';

/**
 * 飞书 API 速率限制延迟（ms）
 */
const RATE_LIMIT_DELAY = 350;

function sleep(ms: number): Promise<void> {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

/**
 * 文档客户端
 * 负责与飞书文档 API 交互
 */
export class DocumentClient {
  constructor(
    private endpoint: string,
    private getAccessToken: () => Promise<string>
  ) {}

  /**
   * 发起飞书 API 请求
   */
  private async request<T>(path: string): Promise<T> {
    await sleep(RATE_LIMIT_DELAY);

    const accessToken = await this.getAccessToken();
    const url = `${this.endpoint}${path}`;

    const response = await fetch(url, {
      headers: {
        Authorization: `Bearer ${accessToken}`,
        'Content-Type': 'application/json',
      },
    });

    if (!response.ok) {
      throw new Error(`API 请求失败: HTTP ${response.status}`);
    }

    const result = (await response.json()) as FeishuApiResponse<T>;

    if (result.code !== 0) {
      throw new Error(`API 错误: [${result.code}] ${result.msg}`);
    }

    return result.data;
  }

  /**
   * 获取 Wiki 节点信息
   * 将 node_token 转换为 document_id
   */
  async getWikiNodeInfo(nodeToken: string): Promise<WikiNodeInfo> {
    return this.request<WikiNodeInfo>(
      `/open-apis/wiki/v2/spaces/get_node?token=${nodeToken}`
    );
  }

  /**
   * 获取文档基本信息
   */
  async getDocumentInfo(documentId: string): Promise<DocumentInfo> {
    return this.request<DocumentInfo>(
      `/open-apis/docx/v1/documents/${documentId}`
    );
  }

  /**
   * 获取文档所有块（自动处理分页）
   */
  async getAllDocumentBlocks(documentId: string): Promise<Block[]> {
    const allBlocks: Block[] = [];
    let pageToken: string | undefined = '';

    do {
      let path = `/open-apis/docx/v1/documents/${documentId}/blocks?document_revision_id=-1&page_size=500`;
      if (pageToken) {
        path += `&page_token=${pageToken}`;
      }

      const data = await this.request<DocumentBlocksResponse>(path);

      if (data.items) {
        allBlocks.push(...data.items);
      }

      pageToken = data.has_more ? data.page_token : undefined;
    } while (pageToken);

    return allBlocks;
  }
}
```

```typescript
import { promises as fs } from 'fs';
import { join } from 'path';
import { homedir } from 'os';
import * as dotenv from 'dotenv';

export interface ConfigProfile {
  appId: string;
  appSecret: string;
  endpoint: string;
  outputDir: string;
}

export interface Config {
  profiles: Record<string, ConfigProfile>;
  currentProfile: string;
}

/**
 * 配置管理器
 */
export class ConfigManager {
  private configPath: string;
  private config: Config;

  constructor(configDir: string = join(homedir(), '.feishu-export')) {
    this.configPath = join(configDir, 'config.json');
    this.config = {
      profiles: {},
      currentProfile: 'default',
    };
  }

  /**
   * 初始化配置
   */
  async init(): Promise<void> {
    // 确保目录存在
    const dir = join(this.configPath, '..');
    await fs.mkdir(dir, { recursive: true });

    // 加载现有配置
    try {
      const content = await fs.readFile(this.configPath, 'utf-8');
      this.config = JSON.parse(content);
    } catch {
      // 配置文件不存在，使用默认值
    }
  }

  /**
   * 获取配置
   */
  async get(key?: string): Promise<any> {
    await this.init();

    if (!key) {
      return this.config;
    }

    const profile = this.config.profiles[this.config.currentProfile];
    return profile?.[key];
  }

  /**
   * 设置配置
   */
  async set(key: string, value: any): Promise<void> {
    await this.init();

    const profile = this.config.profiles[this.config.currentProfile] || {};
    profile[key] = value;

    this.config.profiles[this.config.currentProfile] = profile;
    await this.save();
  }

  /**
   * 保存配置
   */
  private async save(): Promise<void> {
    await fs.writeFile(this.configPath, JSON.stringify(this.config, null, 2));
  }

  /**
   * 重置配置
   */
  async reset(): Promise<void> {
    this.config = {
      profiles: {},
      currentProfile: 'default',
    };
    await this.save();
  }

  /**
   * 获取当前 Profile
   */
  getProfile(name?: string): ConfigProfile {
    const profileName = name || this.config.currentProfile;
    return this.config.profiles[profileName];
  }
}
```

```typescript
import { promises as fs } from 'fs';
import { DocumentClient } from '../core/document.js';
import { MarkdownRenderer } from '../converter/markdown.js';
import { logger } from '../utils/logger.js';

interface DocsOptions {
  file?: string;
  ids?: string;
  output?: string;
  noImages?: boolean;
}

export async function exportDocs(
  documentClient: DocumentClient,
  options: DocsOptions
): Promise<void> {
  let docIds: string[] = [];

  // 从文件读取 ID 列表
  if (options.file) {
    const content = await fs.readFile(options.file, 'utf-8');
    docIds = content
      .split('\n')
      .map((line) => line.trim())
      .filter((line) => line.length > 0 && !line.startsWith('#'));
  }

  // 从命令行读取 ID 列表
  if (options.ids) {
    docIds = options.ids.split(',').map((id) => id.trim());
  }

  if (docIds.length === 0) {
    logger.error('请提供文档 ID 列表（--file 或 --ids）');
    return;
  }

  logger.info(`开始导出 ${docIds.length} 个文档`);

  const renderer = new MarkdownRenderer();
  let successCount = 0;
  let failCount = 0;

  for (let i = 0; i < docIds.length; i++) {
    const docId = docIds[i]!;
    try {
      logger.progress(`[${i + 1}/${docIds.length}] 导出文档: ${docId}`);

      // 获取文档内容
      const docInfo = await documentClient.getDocumentInfo(docId);
      const blocks = await documentClient.getAllDocumentBlocks(docId);

      // 转换 Markdown
      const markdown = renderer.parse(blocks);

      // 写入文件
      const fileName = `${docInfo.document.title}.md`;
      const outputPath = `${options.output}/${fileName}`;
      await fs.writeFile(outputPath, markdown);

      successCount++;
    } catch (err) {
      logger.error(`导出失败: ${docId} - ${err}`);
      failCount++;
    }
  }

  console.log('');
  logger.done(`导出完成: 成功 ${successCount}, 失败 ${failCount}`);
}
```

```typescript
  // }
```

```typescript
/**
 * 生成 Wiki 索引文件
 */
function generateWikiIndex(wiki: WikiNode, basePath: string): string {
  let markdown = `# ${wiki.title}\n\n`;
  markdown += `## 知识库索引\n\n`;

  function renderNode(node: WikiNode, depth: number = 0): string {
    const indent = '  '.repeat(depth);
    const prefix = depth === 0 ? '-' : '*';
    let md = `${indent}${prefix} [${node.title}](${node.title}.md)\n`;

    if (node.children) {
      for (const child of node.children) {
        md += renderNode(child, depth + 1);
      }
    }

    return md;
  }

  markdown += renderNode(wiki);
  return markdown;
}

/**
 * 递归导出 Wiki
 */
async function exportWikiRecursive(
  documentClient: DocumentClient,
  wiki: WikiNode,
  outputDir: string,
  indexOnly: boolean
): Promise<void> {
  const nodeDir = join(outputDir, wiki.title);
  await fs.mkdir(nodeDir, { recursive: true });

  if (wiki.obj_type === 'docx' && !indexOnly) {
    // 导出文档内容
    const blocks = await documentClient.getAllDocumentBlocks(wiki.obj_token);
    const renderer = new MarkdownRenderer();
    const markdown = renderer.parse(blocks);

    await fs.writeFile(join(nodeDir, `${wiki.title}.md`), markdown);
  }

  // 递归处理子节点
  if (wiki.children) {
    for (const child of wiki.children) {
      await exportWikiRecursive(documentClient, child, nodeDir, indexOnly);
    }
  }
}

/**
 * 导出完整 Wiki
 */
export async function exportWiki(
  documentClient: DocumentClient,
  wikiId: string,
  options: WikiOptions
): Promise<void> {
  logger.info('开始导出知识库...');

  // 获取 Wiki 树结构
  const wikiTree = await getWikiTree(documentClient, wikiId);

  // 生成索引文件
  const indexMarkdown = generateWikiIndex(wikiTree, options.output);
  await fs.writeFile(join(options.output, 'README.md'), indexMarkdown);

  if (!options.indexOnly) {
    // 递归导出所有节点
    await exportWikiRecursive(documentClient, wikiTree, options.output, false);
  }

  logger.done(`知识库导出完成: ${options.output}`);
}
```

```typescript
import { promises as fs } from 'fs';
import { join } from 'path';
import type { FileToken } from '../types/index.js';

/**
 * 媒体下载器
 */
export class MediaDownloader {
  constructor(
    private endpoint: string,
    private getAccessToken: () => Promise<string>
  ) {}

  /**
   * 批量下载媒体文件
   */
  async downloadAll(
    fileTokens: FileToken[],
    outputDir: string
  ): Promise<Map<string, string>> {
    const tokenToPath = new Map<string, string>();

    if (fileTokens.length === 0) {
      return tokenToPath;
    }

    const imagesDir = join(outputDir, 'images');
    const filesDir = join(outputDir, 'files');
    await fs.mkdir(imagesDir, { recursive: true });
    await fs.mkdir(filesDir, { recursive: true });

    for (const ft of fileTokens) {
      try {
        const localPath = await this.downloadOne(ft, outputDir, imagesDir, filesDir);
        tokenToPath.set(ft.token, localPath);
      } catch (err) {
        console.error(`下载失败: ${ft.token} - ${err}`);
      }
    }

    return tokenToPath;
  }

  /**
   * 下载单个文件
   */
  private async downloadOne(
    fileToken: FileToken,
    outputDir: string,
    imagesDir: string,
    filesDir: string
  ): Promise<string> {
    const accessToken = await this.getAccessToken();
    const url = `${this.endpoint}/open-apis/drive/v1/medias/${fileToken.token}/download`;

    const response = await fetch(url, {
      headers: {
        Authorization: `Bearer ${accessToken}`,
      },
    });

    if (!response.ok) {
      throw new Error(`HTTP ${response.status}`);
    }

    // 确定保存位置
    const subDir = fileToken.type === 'image' || fileToken.type === 'board' ? 'images' : 'files';
    const dir = subDir === 'images' ? imagesDir : filesDir;

    // 获取文件扩展名
    const contentType = response.headers.get('content-type') || '';
    const ext = this.getExtension(fileToken.type, contentType);

    const fileName = `${fileToken.token}${ext}`;
    const localPath = join(subDir, fileName);
    const fullPath = join(outputDir, localPath);

    // 保存文件
    const buffer = Buffer.from(await response.arrayBuffer());
    await fs.writeFile(fullPath, buffer);

    return localPath;
  }

  /**
   * 获取文件扩展名
   */
  private getExtension(type: FileToken['type'], contentType: string): string {
    if (type === 'image') {
      if (contentType.includes('png')) return '.png';
      if (contentType.includes('gif')) return '.gif';
      if (contentType.includes('webp')) return '.webp';
      return '.jpg';
    }
    if (type === 'board') return '.png';
    return '';
  }
}
```

```typescript
    } catch {

    }
  }

   * 保存元数据

   */
  shouldExport(docId: string, updatedAt: string): boolean {
    const docMeta = this.meta.documents[docId];

    // 首次导出
    if (!docMeta) return true;

    // 检查更新时间
    return new Date(updatedAt) > new Date(docMeta.updatedAt);

    this.meta.documents[docId] = {
      title,
      updatedAt,
      exportedAt: new Date().toISOString(),
      version: (existing?.version || 0) + 1,
```

```typescript
/**
 * 重试处理器
 * 实现指数退避重试策略
 */
export class RetryHandler {
  /**
   * 执行带重试的函数
   */
  async retry<T>(
    fn: () => Promise<T>,
    options: {
      maxRetries?: number;
      backoffMultiplier?: number;
      initialDelay?: number;
    } = {}
  ): Promise<T> {
    const {
      maxRetries = 3,
      backoffMultiplier = 2,
      initialDelay = 1000,
    } = options;

    let lastError: Error | undefined;
    let delay = initialDelay;

    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        return await fn();
      } catch (error) {
        lastError = error as Error;

        if (attempt < maxRetries) {
          console.warn(`重试 ${attempt + 1}/${maxRetries}: ${lastError.message}`);
          await this.sleep(delay);
          delay *= backoffMultiplier;
        }
      }
    }

    throw lastError;
  }

  private sleep(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}
```

### 8.5 故障排除

| 问题 | 解决方案 |
|------|---------|
| 认证失败 | 检查 App ID 和 App Secret 是否正确 |
| 权限不足 | 确认应用已开通必要权限并发布 |
| API 限流 | 使用 `--concurrency` 参数降低并发数 |
| 文档不存在 | 确认文档 ID 正确且有访问权限 |
